HTML 
<div class="bb-rating" aria-label="Rating 5 of 5">
    <span class="bb-rating-label is-checked">
        <span class="bb-rating-icon">star</span>
    </span>
    <span class="bb-rating-label is-checked">
        <span class="bb-rating-icon">star</span>
    </span>
    <!-- ... aynı yapıda diğer öğeler ... -->
</div>
